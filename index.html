// ========== ОБРАБОТКА ОТВЕТОВ (ВСЕ КОМАНДЫ МОГУТ ОТВЕЧАТЬ) ========== // Множество для отслеживания уже ответивших в этом раунде команд let answeredTeams = new Set(); answersRef.on('value', (snapshot) => { const data = snapshot.val(); if(!data || !data.winner || !data.answer) return; if(!waitingForAnswer || !currentQuestion) return; if(timerExpired) return; const team = TEAMS.find(t=>t.id===data.winner); if(!team) return; // Если команда уже отвечала в этом раунде — игнорируем if(answeredTeams.has(team.id)) return; // Помечаем команду как ответившую answeredTeams.add(team.id); const isCorrect = (data.answer === currentQuestion.correct); if(isCorrect) { team.score++; updateScoreboard(); saveScoresToFirebase(); // Показываем на экране ведущего document.getElementById('statusDisplay').innerHTML = `🏆 ${team.name} ответил ПРАВИЛЬНО! +1 очко`; document.getElementById('statusDisplay').style.background = '#2e7d32'; // Подсвечиваем правильный ответ на экране ведущего document.querySelectorAll('#answersPanel .answer-btn').forEach(btn => { if(btn.dataset.answer === currentQuestion.correct) btn.classList.add('correct'); }); // Отправляем подтверждение на телефон команды answersRef.set({ locked: false, winner: data.winner, answer: data.answer, correct: true, message: `${team.name} ответил ПРАВИЛЬНО! +1 очко` }); } else { // Неправильный ответ — показываем, но не начисляем очки const correctAnswerText = currentQuestion.answers[['A','B','C','D'].indexOf(currentQuestion.correct)]; document.getElementById('statusDisplay').innerHTML = `❌ ${team.name} ответил НЕПРАВИЛЬНО! Правильный ответ: ${currentQuestion.correct}. ${correctAnswerText}`; document.getElementById('statusDisplay').style.background = '#b71c1c'; // Подсвечиваем правильный ответ на экране ведущего document.querySelectorAll('#answersPanel .answer-btn').forEach(btn => { if(btn.dataset.answer === currentQuestion.correct) btn.classList.add('correct'); if(btn.dataset.answer === data.answer && data.answer !== currentQuestion.correct) btn.classList.add('wrong'); }); // Отправляем подтверждение на телефон команды answersRef.set({ locked: false, winner: data.winner, answer: data.answer, correct: false, message: `❌ Неправильно! Правильный ответ: ${currentQuestion.correct}` }); } // Через 3 секунды очищаем статус (но не блокируем другие ответы) setTimeout(() => { if(waitingForAnswer) { document.getElementById('statusDisplay').innerHTML = '🟢 Готов к приёму ответов'; document.getElementById('statusDisplay').style.background = '#1a1f2e'; } }, 3000); });
